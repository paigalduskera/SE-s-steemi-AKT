const PDF_FOLDER_ID = '1VQ293v5y7rpu0HezwmxxhNgOrOUb3PVz'; const OBJ_SHEET = 'Objektid'; // Objektid leht: üks grupp = üks rida. Seda tabelit võib käsitsi muuta. // Kui Aadress on tühi, kehtib ülemise rea aadress (mugav käsitsi sisestamisel). // Hooldusleping: "jah" vähemalt ühel objekti real -> objekt on hooldusgraafikus. const OBJ_HEADERS = ['Aadress', 'Objekt', 'Tellija', 'SEJK nr', 'SEJK', 'Grupp', 'SEL', 'SEA', 'HOC', 'iMCP', 'MCP', 'OPUS', 'W&R', 'Hooldusleping']; const COL_CONTRACT = 13; // veerg N (0-põhine indeks) const COUNT_KEYS = ['sel', 'sea', 'hoc', 'imcp', 'mcp', 'opus', 'wr']; // veerud G..M // ---------- Abifunktsioonid ---------- // Aktide leht = alati esimene leht (nagu varem getActiveSheet()) function actsSheet_() { return SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; } // Objektide leht; luuakse automaatselt viimaseks leheks, kui puudub function objSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(OBJ_SHEET); // Varasem katseversioon (JSON-veeruga) nimetatakse ümber, et midagi ei kaoks if (sh && sh.getLastRow() > 0 && sh.getRange(1, 1).getValue() === 'Võti') { sh.setName(OBJ_SHEET + ' (vana)'); sh = null; } if (!sh) { sh = ss.insertSheet(OBJ_SHEET, ss.getSheets().length); sh.appendRow(OBJ_HEADERS); sh.setFrozenRows(1); sh.getRange(1, 1, 1, OBJ_HEADERS.length).setFontWeight('bold'); } else if (sh.getRange(1, COL_CONTRACT + 1).getValue() !== OBJ_HEADERS[COL_CONTRACT]) { // Varasemale lehele lisatakse puuduv veerg "Hooldusleping" sh.getRange(1, COL_CONTRACT + 1).setValue(OBJ_HEADERS[COL_CONTRACT]).setFontWeight('bold'); } return sh; } function objKey_(address) { return String(address || '').toLowerCase().replace(/\s+/g, ' ').trim(); } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function isYes_(v) { return /^(jah|j|x|yes|y|1|true|✓)$/i.test(String(v || '').trim()); } function cellText_(v) { if (v instanceof Date) return Utilities.formatDate(v, 'Europe/Tallinn', 'd.M.yyyy'); return String(v === null || v === undefined ? '' : v).trim(); } // Loeb objektide lehe: tagastab read koos lahendatud aadressiga function readObjRows_() { const sh = objSheet_(); const values = sh.getDataRange().getValues(); const rows = []; let lastAddr = ''; for (let i = 1; i < values.length; i++) { const v = values[i].map(cellText_); if (v.every(x => x === '')) continue; if (v[0]) lastAddr = v[0]; if (!lastAddr) continue; rows.push({ rowIndex: i + 1, address: lastAddr, key: objKey_(lastAddr), v: v }); } return rows; } // Viimane hooldus iga aadressi kohta aktide lehelt function lastMaintenance_() { const sh = actsSheet_(); const values = sh.getDataRange().getValues(); const out = {}; for (let i = 0; i < values.length; i++) { const r = values[i]; const dateStr = cellText_(r[0]); const m = dateStr.match(/(\d{1,2})\.(\d{1,2})\.(\d{4})/); if (!m) continue; // päiserida jms const t = new Date(+m[3], +m[2] - 1, +m[1]).getTime(); const key = objKey_(r[3]); if (!key) continue; if (!out[key] || t >= out[key].t) { out[key] = { t: t, date: m[1] + '.' + m[2] + '.' + m[3], work: cellText_(r[1]), compiler: cellText_(r[5]) }; } } return out; } function listObjects_() { const rows = readObjRows_(); const last = lastMaintenance_(); const byKey = {}; const order = []; rows.forEach(r => { let o = byKey[r.key]; if (!o) { o = byKey[r.key] = { key: r.key, address: r.address, object: '', client: '', contract: false, sejks: [], _sejkIdx: {} }; order.push(r.key); } const v = r.v; if (!o.object && v[1]) o.object = v[1]; if (!o.client && v[2]) o.client = v[2]; if (isYes_(v[COL_CONTRACT])) o.contract = true; const sejkId = (v[3] || '') + '|' + (v[4] || ''); if (!v[3] && !v[4] && !v[5]) return; // ainult päiseinfo rida let sj = o._sejkIdx[sejkId]; if (!sj) { sj = o._sejkIdx[sejkId] = { nr: v[3], name: v[4], grupps: [] }; o.sejks.push(sj); } if (v[5] || COUNT_KEYS.some((k, j) => v[6 + j])) { const g = { name: v[5] }; COUNT_KEYS.forEach((k, j) => { g[k] = v[6 + j]; }); sj.grupps.push(g); } }); return order.map(k => { const o = byKey[k]; delete o._sejkIdx; const l = last[k]; o.lastDate = l ? l.date : ''; o.lastWork = l ? l.work : ''; o.compiler = l ? l.compiler : ''; return o; }); } // Objekti read akti põhjal (kõik read täis aadressiga) function objectRows_(data, contract) { const out = []; (data.sejks || []).forEach((s, i) => { const groups = (s.grupps && s.grupps.length) ? s.grupps : [{}]; const nr = (s.nr !== undefined && s.nr !== null && String(s.nr).trim() !== '') ? String(s.nr).trim() : String(i + 1); groups.forEach(g => { out.push([data.address || '', data.object || '', data.client || '', nr, s.name || '', g.name || ''] .concat(COUNT_KEYS.map(k => (g[k] === undefined || g[k] === null) ? '' : g[k])) .concat([contract || ''])); }); }); if (!out.length) out.push([data.address || '', data.object || '', data.client || '', '', '', ''].concat(COUNT_KEYS.map(() => '')).concat([contract || ''])); return out; } // Asendab objekti read uutega (samas kohas), uus objekt lisatakse lõppu function upsertObject_(data) { const key = objKey_(data.address); if (!key) return null; if (!data.sejks || !data.sejks.length) return key; // struktuurita akti puhul tabelit ei muudeta const sh = objSheet_(); const existing = readObjRows_().filter(r => r.key === key); const old = existing.length ? existing[0].v : null; if (old) { // Säilita käsitsi sisestatud objekti/tellija nimi, kui aktis see puudub data.object = data.object || old[1]; data.client = data.client || old[2]; } // Hooldusleping jääb alles, kui see oli objektil märgitud const contractRow = existing.find(r => isYes_(r.v[COL_CONTRACT])); const rows = objectRows_(data, contractRow ? contractRow.v[COL_CONTRACT] : ''); let at = sh.getLastRow() + 1; if (existing.length) { const first = existing[0].rowIndex; for (let i = existing.length - 1; i >= 0; i--) sh.deleteRow(existing[i].rowIndex); if (first <= sh.getLastRow()) { sh.insertRowsBefore(first, rows.length); at = first; } else at = sh.getLastRow() + 1; } const need = at + rows.length - 1 - sh.getMaxRows(); if (need > 0) sh.insertRowsAfter(sh.getMaxRows(), need); sh.getRange(at, 1, rows.length, OBJ_HEADERS.length).setValues(rows); return key; } // Akti rida aktide lehele (sama veerujärjestus nagu varem) function appendAct_(data) { const sheet = actsSheet_(); const statusMap = { 'k': 'Korras', 'p': 'Puudub', 'o': 'Osaliselt' }; const statusText = statusMap[data.systemStatus] || data.systemStatus || ''; const sejkCount = data.sejks ? data.sejks.length : (data.sejkCount || 0); const rowNumber = sheet.getLastRow() + 1; sheet.appendRow([ data.workDate || new Date().toLocaleDateString('et-EE'), data.workNumber || '', data.object || '', data.address || '', data.client || '', data.compiler || '', data.workType || '', statusText, data.notes || '', sejkCount, data.pdfFilename || '', // PDF link / failinimi new Date().toLocaleString('et-EE') ]); return rowNumber; } // ---------- GET ---------- function doGet(e) { try { const action = e.parameter.action; // Vana vorm (katsetusakt_sheets.html) – töötab edasi nagu enne if (action === 'save') { const data = JSON.parse(e.parameter.data); const rowNumber = appendAct_(data); if (data.sejks && data.sejks.length && data.sejks[0].grupps) upsertObject_(data); return json_({ success: true, message: 'Salvestatud!', rowNumber: rowNumber }); } // Objektide nimekiri koos viimase hooldusega if (action === 'objects') { return json_({ success: true, objects: listObjects_() }); } if (action === 'uploadPDF') { const pdfBase64 = e.parameter.pdfData; const filename = e.parameter.filename || 'Katsetusakt.pdf'; const rowNumber = parseInt(e.parameter.rowNumber); const pdfBlob = Utilities.newBlob(Utilities.base64Decode(pdfBase64.split(',')[1]), 'application/pdf', filename); const folder = DriveApp.getFolderById(PDF_FOLDER_ID); const file = folder.createFile(pdfBlob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); const fileUrl = file.getUrl(); if (rowNumber > 0) actsSheet_().getRange(rowNumber, 11).setValue(fileUrl); return json_({ success: true, message: 'PDF üles laetud!', url: fileUrl }); } return ContentService.createTextOutput('SE Katsetusakt API töötab!'); } catch (error) { Logger.log('Error: ' + error.toString()); return json_({ success: false, message: error.toString() }); } } // ---------- POST (uus vorm saadab terve akti, sest URL-i pikkus on piiratud) ---------- function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(20000); const body = JSON.parse(e.postData.contents); const data = body.data || {}; if (body.action === 'save') { const rowNumber = appendAct_(data); upsertObject_(data); return json_({ success: true, message: 'Salvestatud!', rowNumber: rowNumber }); } // Objekti struktuuri salvestamine ilma aktita if (body.action === 'saveObject') { const key = upsertObject_(data); return json_({ success: !!key, message: key ? 'Objekt salvestatud!' : 'Aadress puudub' }); } return json_({ success: false, message: 'Tundmatu tegevus' }); } catch (error) { Logger.log('Error: ' + error.toString()); return json_({ success: false, message: error.toString() }); } finally { lock.releaseLock(); } }
